cmake_minimum_required(VERSION 3.16)
project(BladeScopeDiagnosisQtTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 REQUIRED COMPONENTS Widgets Sql)
add_executable(diagnosis-qt-test ../bladescope_diagnosis_qt_test.cpp)
target_include_directories(diagnosis-qt-test PRIVATE ../../patches)
target_link_libraries(diagnosis-qt-test PRIVATE Qt5::Widgets Qt5::Sql)
if(MSVC)
  target_compile_options(diagnosis-qt-test PRIVATE /utf-8 /UNDEBUG)
else()
  target_compile_options(diagnosis-qt-test PRIVATE -UNDEBUG)
endif()

add_executable(diagnosis-trial-test ../bladescope_diagnosis_trial_test.cpp)
target_include_directories(diagnosis-trial-test PRIVATE ../../patches)
if(MSVC)
  target_compile_options(diagnosis-trial-test PRIVATE /utf-8 /UNDEBUG)
else()
  target_compile_options(diagnosis-trial-test PRIVATE -UNDEBUG)
endif()
