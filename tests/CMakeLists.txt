cmake_minimum_required(VERSION 3.16)
project(BladeScopeAnalysisTests LANGUAGES CXX)
enable_testing()
foreach(test bladescope_fit_test bladescope_yaw_test)
  add_executable(${test} ${test}.cpp)
  target_compile_features(${test} PRIVATE cxx_std_17)
  target_include_directories(${test} PRIVATE ../patches)
  if(MSVC)
    target_compile_options(${test} PRIVATE /utf-8 /UNDEBUG)
  else()
    target_compile_options(${test} PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ${test} COMMAND ${test})
endforeach()
